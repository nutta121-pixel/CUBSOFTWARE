================================================================
  CUBSOFTWARE SERVER - STARTUP GUIDE
================================================================

  If you are RE-DEPLOYING on an existing VPS, run this first:
  rm -rf ~/cubsoftware
  Then start from STEP 1 below.

================================================================


STEP 1 — Install system prerequisites (fresh VPS only)
-------------------------------------------------------
curl -fsSL https://deb.nodesource.com/setup_lts.x | sudo -E bash -
sudo apt install -y nodejs python3 python3-full python3-pip git nginx
sudo npm install -g pm2


STEP 2 — Add SSH key to GitHub (fresh VPS only)
-------------------------------------------------
ssh-keygen -t ed25519 -C "cubsoftware-vps" -f ~/.ssh/id_ed25519 -N ""
cat ~/.ssh/id_ed25519.pub
# Go to github.com -> Settings -> SSH keys -> New SSH key -> paste the output above
ssh -T git@github.com   # confirm it says: Hi HexEchoTV! You've successfully authenticated


STEP 3 — Clone the repo
------------------------
git clone git@github.com:HexEchoTV/CUBSOFTWARE.git ~/cubsoftware
cd ~/cubsoftware/cubsoftware-server


STEP 4 — Set up the .env file
-------------------------------
cp .env.example .env
nano .env

  WHERE TO GET EACH VALUE:
  -------------------------

  -- Generate these yourself (random strings) --
  FLASK_SECRET_KEY         Run: python3 -c "import secrets; print('CUBSOFTWARE_KEY_' + secrets.token_hex(32))"
                           Used by Flask to sign sessions. Just needs to be a long random string.

  BOT_API_KEY              Run: python3 -c "import secrets; print('CUBSOFTWARE_KEY_' + secrets.token_hex(32))"
                           This is a shared password between the website and the bots.
                           You only set it ONCE here in .env — ecosystem.config.js
                           automatically passes it to each bot under the name they expect:
                             Cub Protector  -> API_KEY
                             CleanMe bot    -> CLEANME_API_KEY
                           If the bot sends the wrong key, the website rejects its requests.

  ADMIN_API_KEY            Run: python3 -c "import secrets; print('CUBSOFTWARE_KEY_' + secrets.token_hex(32))"
                           Used to protect admin-only API endpoints on the website.
                           Only needed if you use those endpoints directly.

  CUBASSIST_ADMIN_KEY      Run: python3 -c "import secrets; print('CUBSOFTWARE_KEY_' + secrets.token_hex(32))"
                           Used to access /cubassist/admin/setup for the Twitch bot OAuth flow.
                           Only needed once during initial Twitch bot setup.

  QUESTCORD_SESSION_SECRET Run: node -e "console.log('CUBSOFTWARE_KEY_' + require('crypto').randomBytes(64).toString('hex'))"
                           Used by QuestCord's web dashboard to sign sessions.

  -- You choose these --
  KERAPLAST_ADMIN_PASSWORD  Whatever password you want for the Keraplast calculator

  -- Discord Developer Portal (discord.com/developers/applications) --
  DISCORD_CLIENT_ID / DISCORD_CLIENT_SECRET
      -> Your cubsoftware.site OAuth app -> OAuth2 -> Client ID / Secret
  CUB_PROTECTOR_TOKEN / CUB_PROTECTOR_CLIENT_ID
      -> Cub Protector app -> Bot -> Token  /  General -> Application ID
  CLEANME_BOT_TOKEN / CLEANME_BOT_CLIENT_ID
      -> CleanMe app -> Bot -> Token  /  General -> Application ID
  QUESTCORD_TOKEN / QUESTCORD_CLIENT_ID / QUESTCORD_CLIENT_SECRET
      -> QuestCord app -> Bot -> Token  /  General -> Application ID  /  OAuth2 -> Secret

  -- Twitch Developer Console (dev.twitch.tv/console) --
  TWITCH_CLIENT_ID / TWITCH_CLIENT_SECRET
      -> Your app -> Manage -> Client ID / Secret

  -- Cloudflare Dashboard (dash.cloudflare.com) --
  CF_API_TOKEN          Profile -> API Tokens -> Create Token (Edit zone DNS permission)
  ZONE_ID_CUBSOFTWARE   cubsoftware.site -> Overview -> right sidebar
  ZONE_ID_QUESTCORD     questcord.fun -> Overview -> right sidebar

  -- Discord Server (right-click server/channel -> Copy ID, needs Developer Mode on) --
  CUB_PROTECTOR_ADMIN_GUILD_ID      Your main Discord server ID
  CUB_PROTECTOR_DEV_GUILD_ID        Your dev/test Discord server ID
  CUB_PROTECTOR_TERMINAL_CHANNEL_ID Channel ID for bot terminal output
  CUB_PROTECTOR_LINKS_LOG_CHANNEL_ID Channel ID for logging links

  -- Anthropic Console (console.anthropic.com) --
  ANTHROPIC_API_KEY     API Keys -> Create Key

  -- Google Cloud Console (console.cloud.google.com) --
  YOUTUBE_API_KEY       APIs & Services -> Credentials -> Create API Key
                        (enable YouTube Data API v3 first)

  -- Buy Me a Coffee (buymeacoffee.com/dashboard) --
  BMAC_TOKEN            Settings -> API

  -- Discord (create a webhook in your server) --
  LINKS_DISCORD_WEBHOOK Server Settings -> Integrations -> Webhooks -> New Webhook -> Copy URL


STEP 5 — Install Python dependencies
--------------------------------------
python3 -m venv apps/cubsoftware-website/venv
apps/cubsoftware-website/venv/bin/pip install -r apps/cubsoftware-website/requirements.txt


STEP 6 — Install Node dependencies
-------------------------------------
cd apps/cub-protector && npm install && cd ../..
cd apps/questcord && npm install && cd ../..
cd apps/cleanme-bot && npm install && cd ../..


STEP 7 — Create data directory
--------------------------------
sudo mkdir -p /var/cubsoftware-data
sudo chown $USER:$USER /var/cubsoftware-data


STEP 8 — Set up Nginx
-----------------------
sudo cp nginx/nginx.conf /etc/nginx/sites-available/cubsoftware
sudo ln -s /etc/nginx/sites-available/cubsoftware /etc/nginx/sites-enabled/cubsoftware
sudo rm -f /etc/nginx/sites-enabled/default
sudo nginx -t
sudo systemctl enable --now nginx


STEP 9 — Start everything with PM2
-------------------------------------
pm2 start ecosystem.config.js
pm2 save
pm2 startup
# Copy and run the command that pm2 startup prints


STEP 10 — Verify everything is running
----------------------------------------
pm2 list
pm2 logs cubsoftware-website --lines 50


STEP 11 — SSL (once DNS is pointed at this VPS)
-------------------------------------------------
sudo apt install -y certbot python3-certbot-nginx
sudo certbot --nginx -d cubsoftware.site -d www.cubsoftware.site
sudo certbot --nginx -d questcord.fun -d www.questcord.fun
sudo systemctl reload nginx


STEP 12 — Cloudflare DDNS auto-update (optional)
--------------------------------------------------
sudo mkdir -p /var/log/cubsoftware
sudo chown $USER:$USER /var/log/cubsoftware
chmod +x cloudflare-ddns.sh
./cloudflare-ddns.sh        # test it manually first
crontab -e
# Add this line to run every 5 minutes:
# */5 * * * * /home/$USER/cubsoftware/cubsoftware-server/cloudflare-ddns.sh


================================================================
  DEPLOYING UPDATES (after first setup)
================================================================

On your local machine:
  git push origin main

On the VPS:
  cd ~/cubsoftware/cubsoftware-server
  git pull origin main && pm2 restart all


================================================================
  USEFUL PM2 COMMANDS
================================================================

pm2 list                        # show all running processes
pm2 logs cubsoftware-website    # live logs for website
pm2 logs cub-protector          # live logs for cub protector
pm2 logs questcord              # live logs for questcord
pm2 logs cleanme-bot            # live logs for cleanme bot
pm2 restart cubsoftware-website # restart website only
pm2 restart all                 # restart everything
